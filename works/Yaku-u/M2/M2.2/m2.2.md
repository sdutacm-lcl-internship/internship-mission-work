### ISO 8601 标准格式    

标准格式    `YYYY-MM-DDTHH:mm:ss.sssZ`   
`Z` 表示 UTC 时间 

```js
new Date().toISOString() // UTC 时间，2025-07-14T13:35:30.123Z
```

转换为东八区

```js
function toISO8601InCN(timestampSecond) {
    // 将时间戳（秒为单位）转化为 Data 对象（毫秒为单位）
    const date = new Date(timestampSecond * 1000);
    //计算东八区的时间pian'yi'li
    const offset = 8 * 60 * 60 * 1000;
    const localDate = new Date(date.getTime() + offset);
    return localDate.toISOString().replace('Z', '+08:00');
}  // 2025-07-14T21:35:30.000+08:00
```

#### dayjs 

```js
import dayjs from 'dayjs';
import utc from 'dayjs/plugin/utc.js';
import timezone from 'dayjs/plugin/timezone.js';

dayjs.extend(utc);
dayjs.extend(timezone);

const timez = 1720966530;
const Time1 = dayjs.unix(timez).tz('Asia/Shanghai').format() 
// 2024-07-14T22:15:30+08:00

// 可在 .format 中自定义格式
const Time2 = dayjs.unix(timez).tz('Asia/Shanghai').format('YYYY-MM-DD HH:mm:ss') 
// 2025-07-14 21:35:30
```
