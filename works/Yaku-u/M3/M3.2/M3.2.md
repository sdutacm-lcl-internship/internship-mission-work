### Express 连接 MySQL

依赖安装

```bash
npm i express mysql2
```

项目结构

```bash
project/
├── db/               # 数据库连接模块
│   └── index.js
├── routes/           # 路由处理
│   └── users.js
├── app.js            # 主入口文件
└── package.json
```

数据库连接模块

```js
//createConnection 只支持单个连接
import mysql from 'mysql2';

const connection = mysql.createConnection({
  host: 'localhost',
  user: 'root',
  password: 'your_password',
  database: 'your_database'
});

connection.connect((err) => {
  if (err) {
    console.error('数据库连接失败:', err);
    return;
  }
  console.log('成功连接数据库');
});

export default connection;
```

配置主文件

```js
const express = require('express');
const mysql = require('mysql2');
const app = express();
const port = 3000;

// 中间件：解析 JSON 请求体
app.use(express.json());

// MySQL 连接配置
const db = mysql.createConnection({
  host: 'localhost', // 数据库主机
  user: 'root',      // MySQL 用户名
  password: 'your_password', // MySQL 密码
  database: 'my_database' // 数据库名称
});

// 连接到 MySQL
db.connect((err) => {
  if (err) {
    console.error('连接 MySQL 失败:', err);
    return;
  }
  console.log('成功连接到 MySQL 数据库');
});

// 测试路由
app.get('/', (req, res) => {
  res.send('Express 服务器运行中！');
});

// 查询数据
app.get('/users', async (req, res) => {
  try {
    const [results] = await db.query('SELECT * FROM users');
    res.json(results); // 返回查询结果
  } catch (err) {
    console.error('Error executing query:', err);
    res.status(500).send('Server error');
  }
});

// 启动服务器
app.listen(port, () => {
  console.log(`服务器运行在 http://localhost:${port}`);
});

```

优化

```js
// 连接池
import mysql from 'mysql2/promise'; //支持 async/await

const pool = mysql.createPool({
    host: '数据库地址',
    user: '用户名',
    password: '密码',
    database: '数据库名',
    waitForConnections: true,
  	connectionLimit: 10, // 最大连接数
  	queueLimit: 0
});

export default pool;
```

>  环境变量：将数据库配置（如用户名、密码）存储在 .env 文件中，使用 dotenv 包加载

### 使用 dotenv 的具体示例

1. 安装 dotenv

   ```bash
   npm install dotenv
   ```

2. 创建 .env 文件（在项目根目录）：

   ```env
   DB_HOST=localhost
   DB_USER=root
   DB_PASSWORD=your_password
   DB_NAME=my_database
   PORT=3000
   ```

3. 加载环境变量

   ```js
   require('dotenv').config();
   const express = require('express');
   const mysql = require('mysql2');
   const app = express();
   
   // 使用环境变量配置 MySQL
   const db = mysql.createConnection({
     host: process.env.DB_HOST,
     user: process.env.DB_USER,
     password: process.env.DB_PASSWORD,
     database: process.env.DB_NAME
   });
   
   // 启动服务器
   app.listen(process.env.PORT, () => {
     console.log(`服务器运行在 http://localhost:${process.env.PORT}`);
   });
   ```
